### Device Instantiation Template
```simics
# Device: [DEVICE_NAME]
instantiate-component component = [device-component]
    name = [device-instance-name]
    base_address = 0x[base-address]
    interrupt_line = [irq-number]
    [additional-parameters]

# Connect device to system bus

```

### Memory Mapping Template
```simics
# Memory map configuration for [platform-name]
create-memory-space name = system.memory

# Map devices to memory space
map [device-instance-name] base = 0x[base-addr] size = 0x[size] space = system.memory

# Map system memory
map system.ram base = 0x[ram-base] size = $memory_size space = system.memory
```

### Interrupt Routing Template
```simics
# Interrupt routing for [platform-name]
create-interrupt-controller name = system.pic

# Connect device interrupts
connect [device-instance-name].interrupt[0] system.pic.interrupt[irq-number]

# Connect interrupt controller to CPU
connect system.pic.cpu_interrupt system.cpu.interrupt
```

## Platform Validation Template

### Validation Checks
```simics
# Platform validation script
script-branch {
    # Wait for platform initialization
    wait-for-time 1.0
    
    # Validate device presence
    @devices = SIM_get_all_objects()
    @expected_devices = ["[device1]", "[device2]", "[device3]"]
    @for device in expected_devices:
    @    assert device in [obj.name for obj in devices], f"Device {device} not found"
    
    # Validate memory map
    @memory_map = SIM_get_object("system.memory").map
    @assert len(memory_map) > 0, "Memory map is empty"
    
    # Validate interrupt routing
    @pic = SIM_get_object("system.pic")
    @assert pic is not None, "Interrupt controller not found"
    
    echo "Platform validation passed"
}
```

## Build and Deployment Template

### Platform Build Script
```bash
#!/bin/bash
# Platform build script for [platform-name]

# Set environment
export SIMICS_HOME=[simics-path]
export PLATFORM_NAME=[platform-name]

# Build device modules
echo "Building device modules..."
for device in [device-list]; do
    echo "Building $device..."
    make -C devices/$device
done

# Generate platform configuration
echo "Generating platform configuration..."
python scripts/generate_platform_config.py --platform $PLATFORM_NAME

# Validate configuration
echo "Validating platform configuration..."
$SIMICS_HOME/bin/simics -no-gui -batch validate_platform.simics

echo "Platform $PLATFORM_NAME build complete"
```
